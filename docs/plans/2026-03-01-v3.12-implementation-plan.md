   ```typescript
   export function humanize(key: string): string {
     return key
       .replace(/([a-z])([A-Z])/g, '$1 $2')  // camelCase → space
       .replace(/[_-]/g, ' ')                  // snake_case/kebab → space
       .replace(/\b\w/g, c => c.toUpperCase()); // capitalize words
   ```
