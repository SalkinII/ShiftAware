### Task 1: Fix Lane Reorder — Shift Y Position Sync

**Files:**
- Modify: `components/features/LaneCalendar/LaneCalendarCanvas.tsx:52-80`

**Step 1: Fix `mergeNodes()` to update shift Y positions**

In `LaneCalendarCanvas.tsx`, replace the `mergeNodes` function (lines 52-80):

```typescript
function mergeNodes(
  currentNodes: Node[],
  laneNodes: Node[],
  newShiftNodes: Node[],
): Node[] {
  const currentShiftMap = new Map<string, Node>();
  for (const node of currentNodes) {
    if (node.id.startsWith("shift-")) {
      currentShiftMap.set(node.id, node);
    }
  }

  const mergedShifts = newShiftNodes.map((newNode) => {
    const existing = currentShiftMap.get(newNode.id);
    if (existing) {
      return {
        ...existing,
        data: newNode.data,
        style: newNode.style,
        position: {
          x: existing.position.x,   // Preserve X (time axis, may be mid-drag)
          y: newNode.position.y,     // Always use calculated Y (lane-determined)
        },
      };
    }
    return newNode;
  });

  return [...laneNodes, ...mergedShifts];
}
```

**Key change:** Line `y: newNode.position.y` replaces the old behaviour of preserving existing Y. Shifts cannot be vertically dragged independently, so the calculated Y from `buildShiftNodes()` is always correct.
